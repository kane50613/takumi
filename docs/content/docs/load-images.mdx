## External Images

In order to make loading more flexible, we do not handle any fetching for you. Instead, exposes the fetched resources via `fetchedResources` option.

If you are using `takumi-js`, it will handle fetching for you, or you can override the default fetching behavior by passing your own `fetchResources` option.

```tsx twoslash tab="takumi-js/node"
/** @jsxImportSource react */
// ---cut-before---
import { fromJsx } from "takumi-js/helpers/jsx";
import { Renderer, extractResourceUrls } from "takumi-js/node";
import { fetchResources } from "takumi-js/helpers";

const renderer = new Renderer();

const { node, stylesheets } = await fromJsx(<img src="https://example.com/image.png" />);

// [!code ++:2]
const urls = extractResourceUrls(node);
const fetchedResources = await fetchResources(urls);

const image = await renderer.render(node, {
  // [!code ++]
  fetchedResources,
  stylesheets,
});
```

```tsx twoslash tab="takumi-js/wasm"
/** @jsxImportSource react */
// ---cut-before---
import { fromJsx } from "takumi-js/helpers/jsx";
import { Renderer, extractResourceUrls } from "takumi-js/wasm";
import { fetchResources } from "takumi-js/helpers";

const renderer = new Renderer();

const { node, stylesheets } = await fromJsx(<img src="https://example.com/image.png" />);

// [!code ++:2]
const urls = extractResourceUrls(node);
const fetchedResources = await fetchResources(urls);

const image = await renderer.render(node, {
  // [!code ++]
  fetchedResources,
  stylesheets,
});
```

## Persistent Images

Preload frequently used images like logo, background, etc. to avoid redundant image decoding and improve performance.

The image key can be used in any `src` field or `background-image`, `mask-image` CSS property.

```tsx twoslash
/** @jsxImportSource react */
// ---cut-before---
import { ImageResponse } from "takumi-js/response";
import type { ImageSource } from "takumi-js/node";

const [logo, background] = await Promise.all([
  fetch("/logo.png").then((res) => res.arrayBuffer()),
  fetch("/background.png").then((res) => res.arrayBuffer()),
]);

const persistentImages: ImageSource[] = [
  {
    src: "my-logo",
    data: logo,
  },
  {
    src: "background",
    data: background,
  },
];

export function GET() {
  return new ImageResponse(<OgImage />, {
    // [!code ++]
    persistentImages,
  });
}

function OgImage() {
  return (
    <div
      style={{
        backgroundImage: "url(background)",
      }}
    >
      <img src="my-logo" />
    </div>
  );
}
```
